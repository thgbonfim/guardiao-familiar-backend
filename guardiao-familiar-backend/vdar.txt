# main.py - Versão Final com Lógica de Filtragem de Remédios Confirmados e Dias da Semana

import os
from datetime import datetime, date, timedelta
from typing import List, Optional
from contextlib import asynccontextmanager
import pytz # Biblioteca para lidar com fusos horários (instale com: pip install pytz)

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, EmailStr
from supabase import create_client, Client
from dotenv import load_dotenv

# --- Configuração Inicial ---
load_dotenv()

# --- Conexão com o Supabase ---
SUPABASE_URL: str = os.getenv("SUPABASE_URL")
SUPABASE_KEY: str = os.getenv("SUPABASE_KEY")

if not SUPABASE_URL or not SUPABASE_KEY:
    raise RuntimeError("Variáveis de ambiente SUPABASE_URL e SUPABASE_KEY não foram definidas.")

supabase: Client = create_client(SUPABASE_URL, SUPABASE_KEY)

# Define o fuso horário de São Paulo
TZ_SAO_PAULO = pytz.timezone('America/Sao_Paulo')

# --- Gerenciador de Ciclo de Vida da Aplicação (Lifespan) ---
@asynccontextmanager
async def lifespan(app: FastAPI):
    print("INFO: Servidor iniciando...")
    print("INFO: Verificando a conexão com o banco de dados Supabase...")
    try:
        response = supabase.table("familias").select("id", count='exact').limit(1).execute()
        if getattr(response, 'error', None):
            raise Exception(f"Erro no Supabase: {response.error.message}")
        print("✅ Conexão com o Supabase bem-sucedida!")
    except Exception as e:
        print("\n" + "!"*60)
        print("❌ ERRO CRÍTICO: Não foi possível conectar ao banco de dados.")
        print(f"   Detalhe: {e}")
        print("   Verifique se as variáveis SUPABASE_URL e SUPABASE_KEY estão corretas no seu arquivo .env")
        print("!"*60 + "\n")
        raise RuntimeError("Falha na inicialização do banco de dados.") from e
    
    yield
    
    print("INFO: Servidor desligado.")


# --- Instância Principal do FastAPI ---
app = FastAPI(
    title="API Guardião Familiar",
    description="API para gerenciar usuários, famílias e lembretes de remédios.",
    version="1.0.3", # Versão incrementada
    lifespan=lifespan
)


# --- Modelos de Dados (Pydantic) para Validação ---
class UsuarioCadastro(BaseModel):
    nome_completo: str
    email: EmailStr
    senha: str

class UsuarioLogin(BaseModel):
    email: EmailStr
    senha: str

class FamiliaCadastro(BaseModel):
    id_do_usuario: str
    nome_da_familia: str

class ParenteCadastro(BaseModel):
    id_da_familia: str
    nome: str
    apelido: Optional[str] = None

class RemedioCadastro(BaseModel):
    id_do_parente: str
    nome_do_remedio: str
    horario: str
    dias_da_semana: List[str]

class RemedioConfirmacao(BaseModel):
    id_do_remedio: str

class FamiliaInfo(BaseModel):
    id: str
    nome_da_familia: str

class ParenteInfo(BaseModel):
    id: str
    nome: str
    apelido: Optional[str] = None
    id_da_familia: str

# Modelo de resposta ajustado para incluir 'dias_da_semana'
class RemedioInfo(BaseModel):
    id: str
    id_do_parente: str
    nome_do_remedio: str
    horario: str
    dias_da_semana: List[str]


# --- Rotas da API (Apenas a última foi alterada) ---
@app.post("/usuarios/login")
async def login_usuario(dados: UsuarioLogin):
    try:
        auth_response = supabase.auth.sign_in_with_password({"email": dados.email, "password": dados.senha})
        user_id = auth_response.user.id
        user_response = supabase.table("usuarios").select("id, nome_completo, id_da_familia").eq("id", user_id).single().execute()
        if not user_response.data: raise HTTPException(status_code=404, detail="Dados do usuário não encontrados no banco.")
        usuario = user_response.data
        familia = None
        if usuario.get('id_da_familia'):
            familia_response = supabase.table("familias").select("id, nome_da_familia").eq("id", usuario['id_da_familia']).single().execute()
            if familia_response.data: familia = familia_response.data
        return {"mensagem": "Login bem-sucedido!", "usuario": usuario, "familia": familia}
    except HTTPException as http_exc: raise http_exc
    except Exception: raise HTTPException(status_code=401, detail="E-mail ou senha inválidos.")

@app.post("/usuarios/cadastrar")
async def cadastrar_usuario(dados: UsuarioCadastro):
    try:
        user_auth = supabase.auth.sign_up({"email": dados.email, "password": dados.senha})
        if not user_auth.user: raise Exception("Falha ao criar usuário na autenticação do Supabase.")
        user_id = user_auth.user.id
        response = supabase.table("usuarios").insert({"id": user_id, "nome_completo": dados.nome_completo, "email": dados.email}).execute()
        if getattr(response, 'error', None):
            supabase.auth.admin.delete_user(user_id)
            raise Exception(response.error.message)
        return {"mensagem": "Usuário cadastrado com sucesso!", "id_do_usuario": user_id}
    except Exception as e: raise HTTPException(status_code=400, detail=str(e))

@app.post("/familias/criar", response_model=FamiliaInfo)
async def criar_familia(dados: FamiliaCadastro):
    try:
        familia_response = supabase.table("familias").insert({"nome_da_familia": dados.nome_da_familia}).execute()
        if getattr(familia_response, 'error', None): raise Exception(familia_response.error.message)
        nova_familia = familia_response.data[0]
        update_response = supabase.table("usuarios").update({"id_da_familia": nova_familia['id']}).eq("id", dados.id_do_usuario).execute()
        if getattr(update_response, 'error', None): raise Exception(update_response.error.message)
        return nova_familia
    except Exception as e: raise HTTPException(status_code=400, detail=str(e))

@app.post("/parentes/cadastrar", response_model=ParenteInfo)
async def cadastrar_parente(dados: ParenteCadastro):
    try:
        response = supabase.table("parentes").insert(dados.model_dump()).execute()
        if getattr(response, 'error', None): raise Exception(response.error.message)
        return response.data[0]
    except Exception as e: raise HTTPException(status_code=400, detail=str(e))

@app.post("/remedios/cadastrar")
async def cadastrar_remedio(dados: RemedioCadastro):
    try:
        response = supabase.table("remedios").insert(dados.model_dump()).execute()
        if getattr(response, 'error', None): raise Exception(response.error.message)
        return {"mensagem": "Lembrete cadastrado!", "remedio": response.data[0]}
    except Exception as e: raise HTTPException(status_code=400, detail=str(e))

@app.post("/remedios/confirmar")
async def confirmar_remedio(dados: RemedioConfirmacao):
    try:
        response = supabase.table("confirmacoes").insert({"id_do_remedio": dados.id_do_remedio}).execute()
        if getattr(response, 'error', None): raise Exception(response.error.message)
        return {"mensagem": "Confirmação registrada com sucesso!"}
    except Exception as e: raise HTTPException(status_code=400, detail=str(e))

@app.get("/familias/{familia_id}/parentes", response_model=List[ParenteInfo])
async def listar_parentes_da_familia(familia_id: str):
    try:
        response = supabase.table("parentes").select("*").eq("id_da_familia", familia_id).execute()
        if getattr(response, 'error', None): raise Exception(response.error.message)
        return response.data
    except Exception as e: raise HTTPException(status_code=500, detail=f"Erro ao buscar parentes: {e}")


# --- ✅ FUNÇÃO REATORADA E CORRIGIDA ---
@app.get("/parentes/{parente_id}/remedios", response_model=List[RemedioInfo])
async def listar_remedios_do_parente(parente_id: str):
    """
    Lista os remédios PENDENTES de um parente para o DIA E DIA DA SEMANA ATUAIS, ordenados por horário.
    """
    try:
        # 1. Obter a data e dia da semana de hoje com fuso horário correto (São Paulo)
        hoje_sp = datetime.now(TZ_SAO_PAULO)
        dia_da_semana_hoje_pt = ["segunda", "terca", "quarta", "quinta", "sexta", "sabado", "domingo"][hoje_sp.weekday()]

        print(f"Buscando remédios para o dia: {dia_da_semana_hoje_pt}")

        # 2. Buscar TODOS os remédios do parente
        todos_remedios_response = supabase.table("remedios") \
            .select("*") \
            .eq("id_do_parente", parente_id) \
            .execute()
        
        if getattr(todos_remedios_response, 'error', None):
            raise Exception(f"Erro ao buscar todos os remédios: {todos_remedios_response.error.message}")

        todos_os_remedios = todos_remedios_response.data
        
        # 3. Filtrar em Python apenas os remédios que são para o dia da semana de HOJE
        remedios_de_hoje = [
            remedio for remedio in todos_os_remedios
            if dia_da_semana_hoje_pt in remedio.get("dias_da_semana", [])
        ]
        
        # Se não há remédios agendados para hoje, retorna lista vazia
        if not remedios_de_hoje:
            return []
            
        # 4. Pegar os IDs dos remédios de hoje para verificar quais já foram confirmados
        ids_remedios_de_hoje = [r['id'] for r in remedios_de_hoje]

        # 5. Buscar os IDs dos remédios já confirmados HOJE
        hoje_inicio_str = hoje_sp.strftime('%Y-%m-%d 00:00:00')
        confirmacoes_response = supabase.table("confirmacoes") \
            .select("id_do_remedio") \
            .in_("id_do_remedio", ids_remedios_de_hoje) \
            .gte("created_at", hoje_inicio_str) \
            .execute()

        if getattr(confirmacoes_response, 'error', None):
            raise Exception(f"Erro ao buscar confirmações: {confirmacoes_response.error.message}")

        ids_confirmados_hoje = {c['id_do_remedio'] for c in confirmacoes_response.data}
        print(f"IDs de remédios já confirmados hoje: {ids_confirmados_hoje}")

        # 6. Filtrar a lista de 'remedios_de_hoje' para remover os já confirmados
        remedios_pendentes = [
            r for r in remedios_de_hoje
            if r['id'] not in ids_confirmados_hoje
        ]
        
        # 7. Ordenar os remédios pendentes por horário
        remedios_pendentes.sort(key=lambda r: r['horario'])

        return remedios_pendentes
        
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Erro no servidor ao buscar remédios: {e}")
